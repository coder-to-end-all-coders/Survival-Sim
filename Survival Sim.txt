def shortNum(number):
    if str(number).find('.') == -1:
        return str(number)
    else:
        return str(number)[:str(number).find('.') + 4]

def addResources(resource, amount):
    global resources
    resources[resource] += amount
    if resource in resourceStorage.keys():
        if resources[resource] > resourceStorage[resource]:
            resources[resource] = resourceStorage[resource]
            bprint('You got '+resource+' but you didn\'t have enough storage for it.')
    bprint('You now have '+shortNum(resources[resource])+' '+resource+'. (+'+shortNum(amount)+')')
    if 'resources' not in unlockedCommands:
        print()
        bprint('You have unlocked the \'resources\' action command.')
        bprint('Use it to see your resources.')
        unlockedCommands.append('resources')
    if resource in foodRef and 'eat' not in unlockedCommands:
        print()
        bprint('You have unlocked the \'eat\' action command.')
        bprint('Use it to eat food.')

def itemEffect(item):
    global gatherEfficiency, exploreEfficiency, unlockedRecipes, resourceStorage
    for effect in itemRef[item]:
        valid = True
        if not effect[0] == 'storage':
            if effect[0] in ['gatherEfficiency']: # add more effects here ______________________________________________
                if item in effectTracker[(effect[0], effect[1])]:
                    valid = False
                else:
                    effectTracker[(effect[0], effect[1])].append(item)
            else: # add more effects here ______________________________________________
                if item in effectTracker[effect[0]]:
                    valid = False
                else:
                    effectTracker[effect[0]].append(item)
        if valid:
            if effect[0] == 'gatherEfficiency':
                gatherEfficiency[effect[1]] += effect[2]
            elif effect[0] == 'exploreEfficiency':
                exploreEfficiency += effect[1]
            elif effect[0] == 'unlockRecipe':
                if effect[1] not in unlockedRecipes:
                    unlockedRecipes.append(effect[1])
            elif effect[0] == 'combat':
                pass # add combat stat when system is done ______________________________________________
            elif effect[0] == 'storage':
                resourceStorage[effect[1]] += effect[2]
            else:
                bprint('I think you forgot something...')

def reverseItemEffect(item):
    global gatherEfficiency, exploreEfficiency, unlockedRecipes, resourceStorage
    for effect in itemRef[item]:
        if effect[0] == 'gatherEfficiency':
            gatherEfficiency[effect[1]] -= effect[2]
        elif effect[0] == 'exploreEfficiency':
            exploreEfficiency -= effect[1]
        elif effect[0] == 'unlockRecipe':
            unlockedRecipes.remove(effect[1])
        elif effect[0] == 'combat':
            pass # add combat stat when system is done ______________________________________________
        elif effect[0] == 'storage':
            resourceStorage[effect[1]] += effect[2]
            if resources[effect[1]] > resourceStorage[effect[1]]:
                resources[effect[1]] = resourceStorage[effect[1]]
        else:
            bprint('I think you forgot something...')

def getItemDesc(item):

            elif effect[0] == 'combat':
                fullDesc += 'unfinished' # add combat stat when system is done ______________________________________________
            elif effect[0] == 'storage':
                fullDesc += 'Will help you store '+effect[1]
            else:
                fullDesc += 'I think you forgot something... '

                fullDesc += ', '
            else:
                fullDesc += '.'

def advanceTime(amount):
    global currentTime, timeDetail, day, year, health, hunger, thirst, sanity, death, possibleDeath, conditions
    currentTime += amount
    while currentTime >= 13:
        currentTime -= 12
        if timeDetail == 'AM':
            timeDetail = 'PM'
        else:
            timeDetail = 'AM'
            day += 1
            while day >= 366:
                day -= 365
                year += 1
    resources['RAW MEAT'] *= .975 * amount # maybe make a better system for food decay ______________________________________________
    hunger -= hungerDecrease * (.9 + random.random() / 5) * amount
    thirst -= thirstDecrease * (.9 + random.random() / 5) * amount
    health -= healthDecrease * (.9 + random.random() / 5) * amount
    if possibleDeath != 'FORGETFULNESS':
        possibleDeath = 'bad health'
    sanity -= 1 * (.9 + random.random() / 5) * amount
    for i in range(amount):
        for condition in conditions.keys():
            if conditions[condition] > 0:
                prevValue = conditions[condition]
                conditions[condition] += random.random() / 5 - .1
                if conditions[condition] < 0:
                    conditions[condition] = 0
                if conditions[condition] >= math.ceil(prevValue):
                    conditionEffect(condition)
                elif conditions[condition] < math.floor(prevValue):
                    reverseConditionEffect(condition)
    if hunger <= 0.0:
        death = 'starvation'
    if thirst <= 0.0:
        death = 'dehydration'
    if health <= 0.0:
        death = possibleDeath
    if sanity <= 0.0:
        sanity = 0.0
    for i in range(amount):
        runWeather()
        if random.random() * 5 < weather - 3.5:
            conditions['hypothermia'] += 1

def runWeather():
    global weather, storm
    canStorm = False
    if weather <= 4:
        canStorm = True
    if weather < 2.5:
        storm = 0
    weather += 1 - (random.random() * 2)
    if storm > 0:
        storm -= 1
    if weather < 0:
        weather = 0
    elif weather > 6:
        weather = 6
    if storm == 0 and weather > 4 and canStorm and random.random() > .75:
        storm = 3 + math.floor(random.random() * 3)

def runTurn():
    global resources, areas, currentArea, health, hunger, thirst, sanity, science, unlockedResearch, displayStatus, textSpeed, statView, conditions
    if displayStatus:
        bprint('Current status: (type \'rebrief\' to see again)')
        print()

        bprint(currentArea)
        if weather < 1:
            bprint('The sky is clear.')
        elif weather < 2:
            bprint('The weather is partly cloudy.')
        elif weather < 3:
            bprint('The sky is covered with clouds.')
        elif weather < 4:
            bprint('A light rain is falling.')
        elif weather < 5:
            bprint('It\'s raining hard.')
        else:
            bprint('It\'s a downpour. Everything is drenched.')
        if storm > 0:
            print('Thunder and lightning fills the air.')
        print()
        bprint(healthMessages[math.floor(health / 10)])
        bprint(hungerMessages[math.floor(hunger / 20)])
        bprint(thirstMessages[math.floor(hunger / 20)])
        print()
    while True:
        bprint('Input action command: (or \'remind\' if you forgot your unlocked actions)')
        action = input('> ')
        print()
        if action == 'gather':
            bprint('Available resources:')
            for resource in areas[currentArea][0].keys():
                if gatherEfficiency[resource] != 0:
                    if resource in resourceStorage.keys():
                        if resourceStorage[resource] > resources[resource]:
                            bprint(resource + ' (~'+shortNum(areas[currentArea][0][resource] * gatherEfficiency[resource])+' per hour of gathering)')
                    else:
                        bprint(resource + ' (~'+shortNum(areas[currentArea][0][resource] * gatherEfficiency[resource])+' per hour of gathering)')
            print()
            bprint('Which resource would you like to gather?')
            choice = ''
            while True:
                choice = input('> ').upper().strip()
                print()
                if choice not in areas[currentArea][0].keys():
                    bprint('That is not a valid resource. Try again.')
                else:
                    if choice in resourceStorage.keys():
                        if resourceStorage[choice] > resources[choice]:
                            break
                        else:
                            bprint('You do not have enough storage.')
                    else:
                        break
            while True:
                bprint('How many hours do you want to gather '+choice+' for?')
                thing = int(input('> '))
                if thing > 0 and thing % 1 == 0:
                    break
                else:
                    bprint('Invalid number. Try again.')
            bprint('You spend '+str(thing)+' hours gathering '+choice+' from the area.')
            increase = areas[currentArea][0][choice] * gatherEfficiency[choice] * thing * (.9 + random.random() / 5)
            addResources(choice, increase)
            print()
            addScience(.1 * (.9 + random.random() / 5) * thing)
            gatherEfficiency[choice] += random.random() / 50
            print()
            break
        elif action == 'explore':
            while True:
                bprint('How many hours do you want to explore for?')
                thing = int(input('> '))
                if thing > 0 and thing % 1 == 0:
                    break
                else:
                    bprint('Invalid number. Try again.')
            bprint('You spend '+str(thing)+' hours exploring the area.')
            print()
            dark = False
            for i in range(thing):
                if (currentTime < 7 and timeDetail == 'AM') or (currentTime > 6 and timeDetail == 'PM'):
                    areas[currentArea][1] += exploreEfficiency * (.9 + random.random() / 5) * .5
                    if not dark:
                        bprint('It\'s dark out, so you have a harder time exploring.')
                        dark = True
                else:
                    areas[currentArea][1] += exploreEfficiency * (.9 + random.random() / 5)
                for location in areas[currentArea][2].keys():
                    if areas[currentArea][2][location][1] == False and areas[currentArea][1] >= areas[currentArea][2][location][0]:
